#include <Servo.h>
#include <Keypad.h>
#include <LiquidCrystal_I2C.h>

#define LED 13
#define BUZZER 2
#define SERVO_PIN 12

// ========================= DETECÇÃO AUTOMÁTICA DO LEITOR DE DIGITAL ============================

#if defined(__has_include)
  #if __has_include(<Adafruit_Fingerprint.h>) && __has_include(<SoftwareSerial.h>)
    #define TEM_LIB_DIGITAL 1
  #endif
#endif

#ifndef TEM_LIB_DIGITAL
  #define TEM_LIB_DIGITAL 0
#endif

#if TEM_LIB_DIGITAL
  #include <Adafruit_Fingerprint.h>
  #include <SoftwareSerial.h>

  SoftwareSerial fingerSerial(3, A0); 
  Adafruit_Fingerprint finger(&fingerSerial);
#endif

#define BTN_DIGITAL A1           

bool sensorOk = false;                
int  simCount = 0;                

// ===================== ESTADOS =====================
enum State {
  CONFIGURAR,
  FECHADO,
  DIGITANDO,
  ABERTO,
  MENU,
  BLOQUEADO,
  ERRO,
  ALERTA
};

State state = FECHADO;

// ===================== PROTÓTIPOS =====================
void enterState(State newState);
void checkPassword();
void resetSistema();
void salvarSenha();
void mostrarAberto();
void voltarAberto();
int  lerDigital();
bool cadastrarDigital();
void limparDigitais();

// ===================== TECLADO =====================
const byte ROWS = 4;
const byte COLS = 4;

char keys[ROWS][COLS] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};

byte rowPins[ROWS] = {11, 10, 9, 8};
byte colPins[COLS] = {7, 6, 5, 4};

Keypad kp = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

// ===================== LCD =====================
LiquidCrystal_I2C lcd(0x27, 16, 2);

// ===================== SENHA =====================
const int PASS_LENGTH = 4;

char correctPass[PASS_LENGTH + 1] = "";
bool senhaDefinida = false;

char pass[PASS_LENGTH + 1];
int i = 0;

// ===================== CONTROLE =====================
int tentativas = 0;
const int MAX_TENT = 3;

unsigned long tempoBloqueio = 0;
const long DURACAO_BLOQUEIO = 15000;

Servo servo;

// ================= LEITOR DE DIGITAL =================

// 0 = nada, 1 = digital reconhecida, -1 = digital desconhecida
int lerDigital() {

#if TEM_LIB_DIGITAL
  if (sensorOk) {

    if (finger.getImage() != FINGERPRINT_OK) return 0;
    if (finger.image2Tz() != FINGERPRINT_OK) return 0;

    uint8_t r = finger.fingerFastSearch();

    if (r == FINGERPRINT_OK) return 1;
    if (r == FINGERPRINT_NOTFOUND) return -1;

    return 0;
  }
#endif

// ==================== SIMULAÇÃO ======================

  if (digitalRead(BTN_DIGITAL) == LOW) {

    delay(50);
    while (digitalRead(BTN_DIGITAL) == LOW);

    return (simCount > 0) ? 1 : -1;
  }

  return 0;
}

#if TEM_LIB_DIGITAL
bool aguardarDedo(bool presente, unsigned long ms) {

  unsigned long t = millis();

  while (millis() - t < ms) {

    uint8_t r = finger.getImage();

    if (presente && r == FINGERPRINT_OK) return true;
    if (!presente && r == FINGERPRINT_NOFINGER) return true;
  }

  return false;
}
#endif

// ================= Cadastra uma NOVA digital =================

bool cadastrarDigital() {

#if TEM_LIB_DIGITAL
  if (sensorOk) {

    finger.getTemplateCount();
    uint16_t id = finger.templateCount + 1;   // próximo ID livre

    lcd.clear();
    lcd.setCursor(1,0);
    lcd.print("Coloque o dedo");

    if (!aguardarDedo(true, 10000)) return false;
    if (finger.image2Tz(1) != FINGERPRINT_OK) return false;

    lcd.clear();
    lcd.setCursor(1,0);
    lcd.print("Tire o dedo");
    delay(1000);

    if (!aguardarDedo(false, 10000)) return false;

    lcd.clear();
    lcd.setCursor(1,0);
    lcd.print("Mesmo dedo");
    lcd.setCursor(1,1);
    lcd.print("novamente");

    if (!aguardarDedo(true, 10000)) return false;
    if (finger.image2Tz(2) != FINGERPRINT_OK) return false;

    if (finger.createModel() != FINGERPRINT_OK) return false;

    return finger.storeModel(id) == FINGERPRINT_OK;
  }
#endif
// =================  SIMULAÇÃO =================

  lcd.clear();
  lcd.setCursor(1,0);
  lcd.print("Aperte o botao");
  lcd.setCursor(1,1);
  lcd.print("p/ cadastrar");

  unsigned long t = millis();

  while (millis() - t < 10000) {

    if (digitalRead(BTN_DIGITAL) == LOW) {

      delay(50);
      while (digitalRead(BTN_DIGITAL) == LOW);

      simCount++;
      return true;
    }
  }

  return false;
}

// ================= APAGA TODAS AS DIGITAIS =================

void limparDigitais() {

#if TEM_LIB_DIGITAL
  if (sensorOk) finger.emptyDatabase();
#endif

  simCount = 0;
}

// ===================== TELAS DO COFRE ABERTO =====================
void mostrarAberto() {

  lcd.clear();

  lcd.setCursor(3,0);
  lcd.print("ACESSO OK");

  lcd.setCursor(0,1);
  lcd.print("C:Fechar *:Menu");
}

// ================= volta para o cofre aberto sem repetir a animação =================

void voltarAberto() {

  state = ABERTO;
  mostrarAberto();
}

// ===================== RESET =====================
void resetSistema() {

  tentativas = 0;
  i = 0;

  noTone(BUZZER);
  digitalWrite(LED, LOW);

  for (int j = 0; j < PASS_LENGTH; j++) {
    pass[j] = '\0';
  }

  servo.write(45);

  lcd.clear();

  lcd.setCursor(1,0);
  lcd.print("RESET GERAL");

  limparDigitais();                   
  lcd.setCursor(0,1);

  for (int k = 0; k < 16; k++) {

    lcd.print((char)255);

    tone(BUZZER, 2500);
    delay(25);
    noTone(BUZZER);
  }

  delay(400);

  enterState(FECHADO);
}

// ===================== SALVAR SENHA =====================
void salvarSenha() {

  for (int j = 0; j < PASS_LENGTH; j++) {
    correctPass[j] = pass[j];
  }
  correctPass[PASS_LENGTH] = '\0';

  senhaDefinida = true;
  tentativas = 0;

  lcd.clear();

  lcd.setCursor(2,0);
  lcd.print("Senha Salva!");

  tone(BUZZER, 2500);
  delay(120);
  tone(BUZZER, 3200);
  delay(120);
  noTone(BUZZER);

  delay(1000);

  enterState(FECHADO);
}

// ===================== ESTADOS =====================

void enterState(State newState) {

  state = newState;

  lcd.clear();
  noTone(BUZZER);

  switch (state) {

    case CONFIGURAR:

      i = 0;

      for (int j = 0; j < PASS_LENGTH; j++) {
        pass[j] = '\0';
      }

      if (!senhaDefinida) {
        lcd.setCursor(1,0);
        lcd.print("Defina a Senha");
      }
      else {
        lcd.setCursor(3,0);
        lcd.print("Nova Senha");
      }

      lcd.setCursor(4,1);
      lcd.print("_ _ _ _");

      break;

    case FECHADO:

      servo.write(90);

      lcd.setCursor(1,0);
      lcd.print("Cofre Fechado");

      lcd.setCursor(1,1);
      lcd.print("Senha/Digital");

      i = 0;

      for (int j = 0; j < PASS_LENGTH; j++) {
        pass[j] = '\0';
      }

      break;

    case DIGITANDO:

      lcd.setCursor(1,0);
      lcd.print("Digite a Senha");

      lcd.setCursor(4,1);
      lcd.print("_ _ _ _");

      break;

    case ABERTO:

      servo.write(0);

      lcd.setCursor(0,0);
      lcd.print("Abrindo Cofre");

      for (int k = 0; k < 16; k++) {

        lcd.setCursor(k,1);
        lcd.print((char)255);

        tone(BUZZER, 1800);
        delay(40);
        noTone(BUZZER);
      }

      mostrarAberto();

      tentativas = 0;

      tone(BUZZER, 2500);
      delay(120);

      tone(BUZZER, 3200);
      delay(120);

      noTone(BUZZER);

      break;

    case MENU:

      lcd.setCursor(1,0);
      lcd.print("Alterar o que?");

      lcd.setCursor(0,1);
      lcd.print("1:Senha 2:Digit.");

      break;

    case BLOQUEADO:

      tempoBloqueio = millis();

      lcd.setCursor(2,0);
      lcd.print("BLOQUEADO");

      lcd.setCursor(2,1);
      lcd.print("Aguarde...");

      break;

    case ALERTA:

      lcd.setCursor(0,0);
      lcd.print("!!! ALERTA !!!");

      break;

    case ERRO:

      lcd.setCursor(1,0);
      lcd.print("Senha Errada");

      lcd.setCursor(3,1);
      lcd.print("Restam:");
      lcd.print(MAX_TENT - tentativas);

      break;
  }
}

// ===================== SENHA =====================
void checkPassword() {

  bool ok = true;

  if (i != PASS_LENGTH) {
    ok = false;
  }

  else {

    for (int j = 0; j < PASS_LENGTH; j++) {

      if (pass[j] != correctPass[j]) {
        ok = false;
      }
    }
  }

  if (ok) {
    enterState(ABERTO);
  }

  else {

    tentativas++;

    if (tentativas >= MAX_TENT) {
      enterState(BLOQUEADO);
    }

    else {
      enterState(ERRO);
    }
  }
}

// ===================== SETUP =====================
void setup() {

  pinMode(LED, OUTPUT);
  pinMode(BUZZER, OUTPUT);
  pinMode(BTN_DIGITAL, INPUT_PULLUP);

#if TEM_LIB_DIGITAL
  finger.begin(57600);
  delay(100);
  sensorOk = finger.verifyPassword();
#endif

  servo.attach(SERVO_PIN);

  lcd.init();
  lcd.backlight();

  // mostra qual leitor está em uso
  lcd.setCursor(0,0);
  lcd.print(sensorOk ? "Leitor: Sensor" : "Leitor: Botao");
  delay(1200);

  servo.write(90);
  enterState(CONFIGURAR);
}

// ===================== LOOP =====================
void loop() {

  char key = kp.getKey();

  // RESET GLOBAL (só depois que existe uma senha)
  if (key == 'B' && senhaDefinida) {
    resetSistema();
    return;
  }

  // SOM TECLAS
  if (key) {

    tone(BUZZER, 3000);
    delay(20);
    noTone(BUZZER);
  }
 // ================= ABRIR COM DIGITAL (cofre fechado ou digitando a senha) =================

  if (state == FECHADO || state == DIGITANDO) {

    int d = lerDigital();

    if (d == 1) {
      enterState(ABERTO);
      return;
    }

    else if (d == -1) {

      tentativas++;

      if (tentativas >= MAX_TENT) enterState(BLOQUEADO);
      else enterState(ERRO);

      return;
    }
  }

  switch (state) {

    // ===================== CONFIGURAR =====================
    case CONFIGURAR:

      if (key) {

        if (key >= '0' && key <= '9') {

          if (i < PASS_LENGTH) {

            int pos = 4 + (i * 2);

            lcd.setCursor(pos,1);
            lcd.print(key);

            delay(300);

            lcd.setCursor(pos,1);
            lcd.print("*");

            pass[i++] = key;
          }
        }

        else if (key == 'D' || key == '#') {

          if (i == PASS_LENGTH) {
            salvarSenha();
          }
        }

        else if (key == '*') {

          if (i > 0) {

            i--;

            pass[i] = '\0';

            int pos = 4 + (i * 2);

            lcd.setCursor(pos,1);
            lcd.print("_");

            lcd.setCursor(pos + 1,1);
            lcd.print(" ");
          }
        }

        else if (key == 'C') {

          for (int j = 0; j < PASS_LENGTH; j++) {
            pass[j] = '\0';
          }

          i = 0;

          lcd.setCursor(4,1);
          lcd.print("_ _ _ _");
        }

        // CANCELAR a troca (só se já existe senha)
        else if (key == 'A' && senhaDefinida) {
          voltarAberto();
        }
      }

      break;

    // ===================== FECHADO =====================
    case FECHADO:

      if (key && key != 'B') {

        if (key == 'A') {
          enterState(ALERTA);
        }

        else {
          enterState(DIGITANDO);
        }
      }

      break;

    // ===================== DIGITANDO =====================
    case DIGITANDO:

      if (key) {

        if (key >= '0' && key <= '9') {

          if (i < PASS_LENGTH) {

            int pos = 4 + (i * 2);

            lcd.setCursor(pos,1);
            lcd.print(key);

            delay(300);

            lcd.setCursor(pos,1);
            lcd.print("*");

            pass[i++] = key;
          }
        }

        else if (key == 'D' || key == '#') {

          checkPassword();
        }

        else if (key == '*') {

          if (i > 0) {

            i--;

            pass[i] = '\0';

            int pos = 4 + (i * 2);

            lcd.setCursor(pos,1);
            lcd.print("_");

            lcd.setCursor(pos + 1,1);
            lcd.print(" ");
          }
        }

        else if (key == 'C') {

          for (int j = 0; j < PASS_LENGTH; j++) {
            pass[j] = '\0';
          }

          i = 0;

          lcd.setCursor(4,1);
          lcd.print("_ _ _ _");
        }
      }

      break;

    // ===================== ABERTO =====================
    case ABERTO:

      if (key == 'C') {
        enterState(FECHADO);
      }

      // MENU DE ALTERAÇÃO (só com o cofre aberto)
      else if (key == '*') {
        enterState(MENU);
      }

      break;

    // ===================== MENU =====================
    case MENU:

      if (key == '1') {
        enterState(CONFIGURAR);          // trocar senha
      }

      else if (key == '2') {

        bool ok = cadastrarDigital();    // cadastrar nova digital

        lcd.clear();
        lcd.setCursor(1,0);
        lcd.print(ok ? "Digital salva!" : "Falhou");

        tone(BUZZER, ok ? 3200 : 800);
        delay(150);
        noTone(BUZZER);

        delay(1200);

        voltarAberto();
      }

      else if (key == 'C') {
        voltarAberto();                  // cancelar
      }

      break;

    // ===================== ERRO =====================
    case ERRO: {

      static unsigned long lastBlink = 0;
      static bool st = false;
      static unsigned long startErro = 0;

      if (startErro == 0) {
        startErro = millis();
      }

      if (millis() - lastBlink > 200) {

        lastBlink = millis();

        st = !st;

        digitalWrite(LED, st);

        if (st) {
          tone(BUZZER, 2000);
        }

        else {
          noTone(BUZZER);
        }
      }

      lcd.setCursor(1,0);
      lcd.print("Senha Errada ");

      lcd.setCursor(3,1);
      lcd.print("Restam:");
      lcd.print(MAX_TENT - tentativas);
      lcd.print(" ");

      if (millis() - startErro > 1000) {

        digitalWrite(LED, LOW);
        noTone(BUZZER);

        for (int j = 0; j < PASS_LENGTH; j++) {
          pass[j] = '\0';
        }

        i = 0;

        startErro = 0;

        enterState(DIGITANDO);
      }

    } break;

    // ===================== BLOQUEADO =====================
    case BLOQUEADO: {

      static unsigned long lastBlink = 0;
      static bool st = false;

      unsigned long tempoPassado = millis() - tempoBloqueio;

      if (tempoPassado >= DURACAO_BLOQUEIO) {

        digitalWrite(LED, LOW);
        noTone(BUZZER);

        lcd.clear();

        lcd.setCursor(2,0);
        lcd.print("Liberado");

        lcd.setCursor(0,1);
        lcd.print("Tente Novamente");

        tone(BUZZER, 2500);
        delay(120);
        noTone(BUZZER);

        delay(1000);

        enterState(FECHADO);

        break;
      }

      if (millis() - lastBlink > 400) {

        lastBlink = millis();

        st = !st;

        digitalWrite(LED, st);

        if (st) {
          tone(BUZZER, 1500);
        }

        else {
          noTone(BUZZER);
        }
      }

      long restante = (DURACAO_BLOQUEIO - tempoPassado) / 1000;

      lcd.setCursor(1,0);
      lcd.print("BLOQUEADO   ");

      lcd.setCursor(2,1);
      lcd.print(restante);
      lcd.print(" segundos ");

    } break;

    // ===================== ALERTA =====================
    case ALERTA: {

      static unsigned long lastBlink = 0;
      static unsigned long lastSiren = 0;

      static bool ledState = false;
      static bool sirenState = false;

      if (millis() - lastBlink > 120) {

        lastBlink = millis();

        ledState = !ledState;

        digitalWrite(LED, ledState);
      }

      if (millis() - lastSiren > 180) {

        lastSiren = millis();

        sirenState = !sirenState;

        if (sirenState) {
          tone(BUZZER, 900);
        }

        else {
          tone(BUZZER, 1500);
        }
      }

      lcd.setCursor(0,0);

      if (ledState)
        lcd.print("!!! ALERTA !!!");
      else
        lcd.print(">>> ALERTA <<<");

      lcd.setCursor(1,1);
      lcd.print("A=desligar");

      if (key == 'A') {

        digitalWrite(LED, LOW);

        noTone(BUZZER);

        lcd.clear();

        lcd.setCursor(2,0);
        lcd.print("Alerta OFF");

        delay(800);

        enterState(FECHADO);
      }

    } break;
  }
}
